name: APK bauen
on:
  workflow_dispatch:
  push:
    branches: [main, master]
permissions:
  contents: write
jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - name: Abhaengigkeiten installieren
        run: npm install
      - name: Spiel bauen (dist/index.html)
        run: node build.mjs
      - name: www-Ordner fuer Capacitor
        run: node make_www.mjs
      - name: Android-Projekt erzeugen
        run: |
          npx cap add android
          python3 patch_android.py android
          npx cap sync android
      - name: APK bauen
        working-directory: android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon
      - name: APK hochladen
        uses: actions/upload-artifact@v4
        with:
          name: gemen3d-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
      - name: APK als Release veroeffentlichen (direkter Download am Handy)
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Gemen 3D Build ${{ github.run_number }}
          files: android/app/build/outputs/apk/debug/app-debug.apk
